#!/usr/bin/env python3

import argparse
import json
import os
import re

from urllib.parse import urlparse


SHORTENERS = {

    "bit.ly",
    "t.co",
    "tinyurl.com",
    "goo.gl",
    "ow.ly",
    "is.gd",
    "buff.ly",
    "cutt.ly"
}


def main():

    parser = argparse.ArgumentParser(
        description="GxURLShield - análisis local de URLs"
    )

    parser.add_argument(
        "url",
        help="URL que deseas analizar"
    )

    parser.add_argument(
        "-o",
        "--output",
        default="reports/urlshield.json"
    )

    args = parser.parse_args()

    url = args.url.strip()

    parsed = urlparse(url)

    host = (
        parsed.hostname or ""
    ).lower()

    findings = []

    # Esquema

    if parsed.scheme not in (
        "http",
        "https"
    ):

        findings.append(
            "Esquema no HTTP/HTTPS"
        )

    # Acortador

    if host in SHORTENERS:

        findings.append(
            "URL utiliza un acortador conocido"
        )

    # IP literal

    if re.fullmatch(
        r"\d{1,3}(?:\.\d{1,3}){3}",
        host
    ):

        findings.append(
            "El host es una dirección IP"
        )

    # @

    if "@" in parsed.netloc:

        findings.append(
            "La autoridad contiene @"
        )

    # Credenciales

    if parsed.username or parsed.password:

        findings.append(
            "La URL contiene credenciales embebidas"
        )

    # CR/LF

    if "\r" in url or "\n" in url:

        findings.append(
            "La URL contiene caracteres CR/LF"
        )

    # Longitud

    if len(url) > 2048:

        findings.append(
            "URL extremadamente larga"
        )

    risk = (
        "review"
        if findings
        else "low"
    )

    report = {

        "tool": "GxURLShield",

        "version": "1.1.0",

        "url": url,

        "host": host,

        "scheme":
            parsed.scheme,

        "findings":
            findings,

        "risk":
            risk
    }

    os.makedirs(
        os.path.dirname(args.output) or ".",
        exist_ok=True
    )

    with open(
        args.output,
        "w",
        encoding="utf-8"
    ) as file:

        json.dump(
            report,
            file,
            indent=2,
            ensure_ascii=False
        )

    print("\n[+] GxURLShield")

    print(
        f"[+] Host: {host}"
    )

    print(
        f"[+] Hallazgos: {len(findings)}"
    )

    print(
        f"[+] Riesgo: {risk}"
    )

    print(
        f"[+] Reporte: {args.output}"
    )

    main()
